#!/usr/bin/env python3
"""Rebuild the OfficePress icon set (Lucide). Python 3.9+, stdlib only.

Usage:
  python3 scripts/build_icons.py                    # rebuild sprite + js/icons.js from icons/svg
  python3 scripts/build_icons.py --add truck receipt # download official Lucide icons, then rebuild
Rules: Lucide only (one library per surface). Names are Lucide's kebab-case names
(browse https://lucide.dev/icons). Pinned to the version in icons/VERSION.
"""
import json, os, re, sys, urllib.request

KIT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
SVG_DIR = os.path.join(KIT, "icons", "svg")

def version():
    return open(os.path.join(KIT, "icons", "VERSION")).read().strip()

def download(names):
    ver = version()
    for n in names:
        url = f"https://unpkg.com/lucide-static@{ver}/icons/{n}.svg"
        try:
            data = urllib.request.urlopen(url, timeout=20).read().decode()
        except Exception as e:
            print(f"✗ {n}: {e} — check the name at https://lucide.dev/icons"); continue
        open(os.path.join(SVG_DIR, n + ".svg"), "w").write(data)
        print(f"✓ {n}")

def build():
    ver = version(); syms = []; names = []
    for f in sorted(os.listdir(SVG_DIR)):
        if not f.endswith(".svg"): continue
        n = f[:-4]
        body = re.search(r"<svg[^>]*>(.*)</svg>", open(os.path.join(SVG_DIR, f)).read(), re.S).group(1)
        body = re.sub(r"\s+", " ", body).strip().replace(" />", "/>")
        syms.append(f'<symbol id="i-{n}" viewBox="0 0 24 24">{body}</symbol>'); names.append(n)
    open(os.path.join(KIT, "icons", "sprite.svg"), "w").write(
        '<svg xmlns="http://www.w3.org/2000/svg">\n<!-- Lucide v%s (ISC) — OfficePress icon set. Symbol ids: i-<name> -->\n%s\n</svg>\n' % (ver, "\n".join(syms)))
    js = '''/* OfficePress icons — Lucide v%s (ISC). Generated by scripts/build_icons.py — do not edit.
   Inlines the sprite so <use href="#i-name"> works from file:// and any server.
   Usage: <svg class="op-icon" aria-hidden="true"><use href="#i-bell"/></svg> */
(function () {
  var SPRITE = %s;
  function inject() {
    if (document.getElementById("op-icon-sprite")) return;
    var host = document.createElement("div");
    host.innerHTML = '<svg id="op-icon-sprite" xmlns="http://www.w3.org/2000/svg" style="position:absolute;width:0;height:0;overflow:hidden" aria-hidden="true">' + SPRITE + "</svg>";
    document.body.insertBefore(host.firstChild, document.body.firstChild);
  }
  window.OP_ICON_NAMES = %s;
  if (document.body) inject(); else document.addEventListener("DOMContentLoaded", inject);
})();
''' % (ver, json.dumps("".join(syms)), json.dumps(names))
    open(os.path.join(KIT, "js", "icons.js"), "w").write(js)
    json.dump({"library": "lucide", "version": ver, "license": "ISC", "prefix": "i-", "icons": names},
              open(os.path.join(KIT, "icons", "icons.json"), "w"), indent=2)
    print(f"Built {len(names)} icons (Lucide {ver})")

if __name__ == "__main__":
    args = sys.argv[1:]
    if args and args[0] == "--add":
        download(args[1:])
    build()
